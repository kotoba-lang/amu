# below). 25/28/29/30/31 are NOT admissible at this grid (33,696 lacks the

run() { # cycle cell W lanes chunks
  log="$EV/run-$1-$2.log"

}

# cycle 1: forward
run 1 e24 64 2156544 24
run 1 e26 64 2156544 26
run 1 e27 64 2156544 27
run 1 e32 64 2156544 32
# cycle 2: reversed
run 2 e32 64 2156544 32
run 2 e27 64 2156544 27
run 2 e26 64 2156544 26

# cycle 3: rotated
run 3 e26 64 2156544 26
run 3 e27 64 2156544 27
