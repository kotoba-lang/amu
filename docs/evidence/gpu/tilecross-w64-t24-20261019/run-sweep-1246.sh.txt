#!/bin/sh
# NOTE (repo rule): kept as run-sweep-1246.sh.txt because the repository workflow-lint
# refuses POSIX shell execution files; run with sh <this file>.
# W=64 TILE-CROSSING BISECTION AT THE NEAREST COMMON-ADMISSIBLE LANE COUNT
# (JUN-58, tick 20261019-1246): grid B, W=64 fixed.
#
# tick-20261019-1217 (PR #1359) left the crossing in 17..31 tiles and named
# "tiles 24 (L=1536)" as the next bisection -- but at lanes 2^21 that shape
# is NOT admissible: valid-shape? needs lanes/(W*chunks) integral, and at
# 2^21 only tiles dividing 2^15 (powers of two) qualify. 17..31 contains
# none. Demonstrated in refuse-t24-at-2pow21.log (exit 3).
#
# Design decision for this slice: move grid B to lanes 2,095,104 =
# 6144*341, the nearest lane count to 2^21 where tiles 8/16/24/32 (and 1)
# are all admissible (wg = 4092/2046/1364/1023; lanes/256 = 8184; the
# L=2048 f32-exact bound 14,681,088 < 2^24 holds). New cell is n24.
# Calibration cell cal16 runs the PR-#1359 t16 cell exactly (W=64,
# lanes=2^21, chunks=16) in the same session to bound session drift.
# 3 cycles, cell order rotated (forward, reversed, rotated).
# Idempotent: a cell whose log already exists is skipped.
set -e
TAB=$(printf '\t')
cd "$(dirname "$0")/../../../.."
EV=docs/evidence/gpu/tilecross-w64-t24-20261019
mkdir -p "$EV"

run() { # cycle cell W lanes chunks
  log="$EV/run-$1-$2.log"
  if [ -f "$log" ] && grep -q "RESULT${TAB}verified=true" "$log"; then echo "skip $log"; return 0; fi
  echo "run $log"
  (date; uptime; time kbb --backend sci scripts/test-gpu-subgroup-reduce-chunksweep.cljk "$3" "$4" "$5") > "$log" 2>&1
  grep -q "RESULT	verified=true" "$log" || { echo "UNVERIFIED $log"; exit 1; }
}

# new grid: lanes 2095104 ; calibration: lanes 2097152
# cycle 1: forward
run 1 cal16 64 2097152 16
run 1 n8    64 2095104 8
run 1 n16   64 2095104 16
run 1 n24   64 2095104 24
run 1 n32   64 2095104 32
# cycle 2: reversed
run 2 n32   64 2095104 32
run 2 n24   64 2095104 24
run 2 n16   64 2095104 16
run 2 n8    64 2095104 8
run 2 cal16 64 2097152 16
# cycle 3: rotated
run 3 n16   64 2095104 16
run 3 n24   64 2095104 24
run 3 n32   64 2095104 32
run 3 cal16 64 2097152 16
run 3 n8    64 2095104 8
echo SWEEP-COMPLETE
